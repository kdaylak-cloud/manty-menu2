const CATEGORIES = ['Tümü', 'Yöresel Mantılar', 'Orta Asya Mantıları', 'Modern Yorum Mantılar', 'Uzak Doğu Mantıları', 'İçecekler'];
const apiKey = ""; 

export default function App() {
  const [activeCategory, setActiveCategory] = useState('Tümü');
  const [searchQuery, setSearchQuery] = useState('');
  const [isAIChatOpen, setIsAIChatOpen] = useState(false);
  const [isQRModalOpen, setIsQRModalOpen] = useState(false);
  const [chatMessages, setChatMessages] = useState([
    { role: 'assistant', text: 'Merhaba! Ben Manty\'nin Gurme Danışmanı. Bugün size nasıl yardımcı olabilirim?' }
  ]);
  const [userInput, setUserInput] = useState('');
  const [isAILoading, setIsAILoading] = useState(false);
  const chatEndRef = useRef(null);

  const qrUrl = `https://api.qrserver.com/v1/create-qr-code/?size=300x300&data=${encodeURIComponent(window.location.href)}`;

  const scrollToBottom = () => {
    chatEndRef.current?.scrollIntoView({ behavior: "smooth" });
  };

  useEffect(() => {
    scrollToBottom();
  }, [chatMessages]);

  const callGeminiAI = async (userQuery) => {
    setIsAILoading(true);
    const menuContext = MOCK_DATA.map(item => `${item.name}: ${item.description}`).join('\n');
    const systemPrompt = `Sen Manty restoranının garsonusun. Menü: ${menuContext}`;

    try {
      const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          contents: [{ parts: [{ text: userQuery }] }],
          systemInstruction: { parts: [{ text: systemPrompt }] }
        })
      });
      const data = await response.json();
      const aiText = data.candidates?.[0]?.content?.parts?.[0]?.text;
      setChatMessages(prev => [...prev, { role: 'assistant', text: aiText || "Bir hata oluştu." }]);
    } catch (error) {
      setChatMessages(prev => [...prev, { role: 'assistant', text: "Şu an cevap veremiyorum." }]);
    } finally {
      setIsAILoading(false);
    }
  };

  const handleSendMessage = (e) => {
    e.preventDefault();
    if (!userInput.trim()) return;
    setChatMessages(prev => [...prev, { role: 'user', text: userInput }]);
    callGeminiAI(userInput);
    setUserInput('');
  };

  const filteredItems = MOCK_DATA.filter((item) => {
    const matchesCategory = activeCategory === 'Tümü' || item.category === activeCategory;
    const matchesSearch = item.name.toLowerCase().includes(searchQuery.toLowerCase());
    return matchesCategory && matchesSearch;
  });

  return (
    <div className="min-h-screen bg-[#fdf9f1] text-stone-800 font-sans pb-20 relative">
      {/* Üst Alan */}
      <header className="bg-stone-900 shadow-lg rounded-b-[2.5rem] relative overflow-hidden">
        <div className="relative pt-16 pb-10 px-6 z-10 flex flex-col items-center text-center text-white">
          <h1 className="text-6xl font-bold mb-2" style={{ fontFamily: 'Georgia, serif' }}>Manty</h1>
          <p className="text-amber-400 text-xs tracking-widest uppercase mb-8">Mantı Restoranı</p>
          <div className="relative w-full max-w-md">
            <Search className="absolute left-4 top-3.5 h-5 w-5 text-stone-500" />
            <input
              type="text"
              className="w-full pl-12 pr-4 py-3.5 rounded-2xl bg-white text-stone-900 outline-none"
              placeholder="Lezzet arayın..."
              value={searchQuery}
              onChange={(e) => setSearchQuery(e.target.value)}
            />
          </div>
        </div>
      </header>

      {/* Kategoriler */}
      <div className="sticky top-0 z-20 bg-[#fdf9f1]/95 backdrop-blur-md border-b border-[#ebdcca] py-4 px-4 flex overflow-x-auto gap-2 scrollbar-hide">
        {CATEGORIES.map((cat) => (
          <button
            key={cat}
            onClick={() => setActiveCategory(cat)}
            className={`whitespace-nowrap px-6 py-2.5 rounded-full text-sm font-semibold ${activeCategory === cat ? 'bg-amber-600 text-white shadow-lg' : 'bg-white text-stone-600 border border-[#ebdcca]'}`}
          >
            {cat}
          </button>
        ))}
      </div>

      {/* Menü Listesi */}
      <main className="px-4 py-8 max-w-3xl mx-auto space-y-8">
        {filteredItems.map((item) => (
          <div key={item.id} className="bg-white rounded-[2rem] shadow-sm border border-stone-100 overflow-hidden flex flex-col sm:flex-row h-auto">
            <img src={item.image} alt={item.name} className="w-full sm:w-2/5 h-56 sm:h-auto object-cover" />
            <div className="p-6 sm:p-8 flex-1">
              <div className="flex justify-between mb-3">
                <h3 className="text-2xl font-bold text-stone-900">{item.name}</h3>
                <span className="text-xl font-black text-amber-600">{item.price}</span>
              </div>
              <p className="text-stone-500 text-sm whitespace-pre-line mb-4">{item.description}</p>
              <div className="pt-4 border-t border-stone-50 flex justify-between items-center text-xs font-bold text-stone-400 uppercase tracking-widest">
                <span>{item.category}</span>
                <ChevronRight className="w-5 h-5 text-amber-600" />
              </div>
            </div>
          </div>
        ))}
      </main>

      {/* Floating Buttons */}
      <div className="fixed bottom-6 right-6 z-50 flex flex-col gap-4">
        <button onClick={() => setIsQRModalOpen(true)} className="bg-white p-4 rounded-full shadow-2xl border border-stone-200"><QrCode className="w-6 h-6" /></button>
        <button onClick={() => setIsAIChatOpen(true)} className="bg-stone-900 text-white p-4 rounded-full shadow-2xl flex items-center gap-2 group">
          <span className="max-w-0 overflow-hidden whitespace-nowrap group-hover:max-w-xs transition-all font-bold text-sm">Gurme Danışman ✨</span>
          <MessageSquare className="w-6 h-6" />
        </button>
      </div>

      {/* QR Modal */}
      {isQRModalOpen && (
        <div className="fixed inset-0 z-[70] bg-black/80 flex items-center justify-center p-6" onClick={() => setIsQRModalOpen(false)}>
          <div className="bg-white w-full max-w-sm rounded-[3rem] p-8 flex flex-col items-center" onClick={e => e.stopPropagation()}>
            <div className="flex justify-between w-full mb-6">
              <h3 className="text-xl font-bold">Masaüstü QR Kod</h3>
              <X className="w-6 h-6 cursor-pointer" onClick={() => setIsQRModalOpen(false)} />
            </div>
            <div className="bg-[#fdf9f1] p-6 rounded-[2rem] border-4 border-stone-900 mb-6">
              <img src={qrUrl} alt="QR" className="w-48 h-48" />
            </div>
            <button onClick={() => window.print()} className="w-full flex items-center justify-center gap-2 bg-stone-900 text-white py-4 rounded-2xl font-bold"><Printer className="w-5 h-5" /> Yazdır</button>
          </div>
        </div>
      )}

      {/* AI Chat Modal */}
      {isAIChatOpen && (
        <div className="fixed inset-0 z-[60] bg-black/60 backdrop-blur-sm flex items-end sm:items-center justify-center p-0 sm:p-4">
          <div className="bg-[#fdf9f1] w-full max-w-lg h-[80vh] sm:rounded-[2.5rem] flex flex-col shadow-2xl overflow-hidden">
            <div className="bg-stone-900 p-6 flex justify-between items-center text-white">
              <div className="flex items-center gap-3">
                <Sparkles className="w-6 h-6 text-amber-500" />
                <div><h2 className="font-bold">Gurme Danışman</h2><p className="text-xs text-stone-400">Yapay Zeka Önerileri</p></div>
              </div>
              <X className="w-6 h-6 cursor-pointer" onClick={() => setIsAIChatOpen(false)} />
            </div>
            <div className="flex-1 overflow-y-auto p-6 space-y-4">
              {chatMessages.map((msg, i) => (
                <div key={i} className={`flex ${msg.role === 'user' ? 'justify-end' : 'justify-start'}`}>
                  <div className={`max-w-[85%] p-4 rounded-2xl text-sm ${msg.role === 'user' ? 'bg-stone-900 text-white rounded-tr-none' : 'bg-white border rounded-tl-none'}`}>{msg.text}</div>
                </div>
              ))}
              {isAILoading && <div className="flex gap-2 items-center text-xs text-stone-500"><Loader2 className="w-4 h-4 animate-spin" /> Bekleyiniz...</div>}
              <div ref={chatEndRef} />
            </div>
            <form onSubmit={handleSendMessage} className="p-6 bg-white border-t flex gap-2">
              <input type="text" value={userInput} onChange={e => setUserInput(e.target.value)} className="flex-1 bg-stone-50 rounded-2xl p-4 text-sm outline-none" placeholder="Bugün ne önerirsin?" />
              <button type="submit" className="bg-stone-900 text-white p-4 rounded-xl"><Send className="w-5 h-5" /></button>
            </form>
          </div>
        </div>
      )}
    </div>
  );
}
